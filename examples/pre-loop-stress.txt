// Twelve random variables irrelevant to the loop, in front of the six-bit
// counter loop of stress-test-6. Benchmarks the cone-of-influence projection:
// all 2^12 loop entries share one projected loop kernel.
z1 ~ bernoulli 0.5;
z2 ~ bernoulli 0.5;
z3 ~ bernoulli 0.5;
z4 ~ bernoulli 0.5;
z5 ~ bernoulli 0.5;
z6 ~ bernoulli 0.5;
z7 ~ bernoulli 0.5;
z8 ~ bernoulli 0.5;
z9 ~ bernoulli 0.5;
z10 ~ bernoulli 0.5;
z11 ~ bernoulli 0.5;
z12 ~ bernoulli 0.5;
timesBit0 := false;
timesBit1 := false;
timesBit2 := false;
timesBit3 := false;
timesBit4 := false;
timesBit5 := false;
overflow := false;

do {
  coin ~ bernoulli 0.5;
  if not timesBit0 then {
    timesBit0 := true;
  } else if not timesBit1 then {
      timesBit1 := true;
      timesBit0 := false;
  } else if not timesBit2 then {
      timesBit2 := true;
      timesBit0 := false;
      timesBit1 := false;
  } else if not timesBit3 then {
      timesBit3 := true;
      timesBit0 := false;
      timesBit1 := false;
      timesBit2 := false;
  } else if not timesBit4 then {
      timesBit4 := true;
      timesBit0 := false;
      timesBit1 := false;
      timesBit2 := false;
      timesBit3 := false;
  } else if not timesBit5 then {
      timesBit5 := true;
      timesBit0 := false;
      timesBit1 := false;
      timesBit2 := false;
      timesBit3 := false;
      timesBit4 := false;
  } else {
      overflow := true;
  }
} while coin;

return overflow;
